# Faz 4 — SAP & OTP Entegrasyonu (Kritik Yol)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement SAP dual-channel integration (GUI Scripting + NCo) and OTP/2FA module with 5 channel implementations, enabling "portal login + SAP transaction" workflows with human-in-the-loop approval fallback.

**Architecture:** 
- **SAP Strategy:** Hybrid channels — ISapGuiChannel (UI Scripting, fallback) and ISapDataChannel (NCo, primary). Engine routes activity execution based on explicit channel choice in workflow JSON.
- **OTP Pattern:** IOtpChannel interface with 5 implementations (Email, TOTP, GsmModem, PhoneForward, HumanApproval) + fallback chain support. GetOtp activity queries channel sequentially; timeout → BusinessException or next channel.
- **UI Spy:** Attended agent module using Windows API (user32.dll) + SAP COM to extract element hierarchy; results streamed to Studio via SignalR for workflow designer integration.

**Tech Stack:** SAP GUI Scripting COM, SAP NCo 3.1, Otp.NET (TOTP), MailKit (Email OTP), user32 P/Invoke, SignalR client, SecureString (credential vault).

## Global Constraints

- SAP GUI Scripting & NCo implementations **Windows-only**, decorated with `[SupportedOSPlatform("windows")]`.
- Credential handling: System/Client/User/Language from Vault via `ICredentialVault`, never plaintext in logs or config.
- OTP codes stored encrypted in database with short TTL (typically 5 min); masked in logs ("OTP: [MASKED]").
- OTP timeout default 3 minutes (spec Bölüm 7); configurable per OtpRequest.
- SAP channel selection explicit in workflow JSON node property `"channel": "gui" | "nco"` (spec Bölüm 5.1).
- All 5 OTP channels tested with mocks; Email/TOTP/GsmModem require integration test fixtures (spec Bölüm 12).
- Code coverage: Infrastructure ≥70%, WebAPI ≥65%.
- TDD mandatory: failing test → minimal impl → pass → commit per step.

```csharp
// src/RPA.Infrastructure/SAP/SapConnectionPool.cs
[SupportedOSPlatform("windows")]
public class SapConnectionPool : IDisposable
{
    private readonly ConcurrentDictionary<string, RfcDestination> _pool;
    private readonly ICredentialVault _vault;
    private readonly ILogger<SapConnectionPool> _logger;

    public SapConnectionPool(ICredentialVault vault, ILogger<SapConnectionPool> logger)
    {
        _pool = new ConcurrentDictionary<string, RfcDestination>();
        _vault = vault;
        _logger = logger;
    }

    public async Task<RfcDestination> GetConnectionAsync(SapConnection connection, CancellationToken cancellationToken)
    {
        string key = $"{connection.SystemName}_{connection.Client}";
        
        if (_pool.TryGetValue(key, out var dest))
        {
            _logger.LogDebug("Reusing SAP connection {Key}", key);
            return dest;
        }

        _logger.LogInformation("Creating new SAP NCo connection {Key}", key);
        // TODO: Call RfcDestinationManager with cred from vault
        var newDest = new RfcDestination(); // Placeholder
        _pool.TryAdd(key, newDest);
        return newDest;
    }

    public void ReturnConnection(string key, RfcDestination dest)
    {
        _logger.LogDebug("Returning connection {Key}", key);
        // Keep in pool for reuse
    }

    public void Dispose()
    {
        foreach (var dest in _pool.Values)
        {
            dest?.Dispose();
        }
        _pool.Clear();
    }
}
```

```csharp
// src/RPA.Infrastructure/Activities/GetOtpActivity.cs
[Activity(
    Id = "Otp.GetOtp",
    Name = "Get OTP Code",
    Description = "Retrieve OTP from specified channel(s) with fallback support",
    Category = "Integration")]
public class GetOtpActivity : IActivity
{
    private readonly OtpChannelFactory _factory;
    private readonly RpaDbContext _db;
    private readonly ILogger<GetOtpActivity> _logger;

    public GetOtpActivity(OtpChannelFactory factory, RpaDbContext db, ILogger<GetOtpActivity> logger)
    {
        _factory = factory;
        _db = db;
        _logger = logger;
    }

    public async Task<ActivityResult> ExecuteAsync(ActivityContext context)
    {
        var channels = (List<OtpChannelType>)context.GetInput("channels");
        var portal = (string)context.GetInput("portalReference");
        var timeout = TimeSpan.FromSeconds((int?)context.GetInput("timeoutSeconds") ?? 180);

        var otpRequest = new OtpRequest
        {
            JobRunId = context.JobRunId,
            Channel = channels[0],
            PortalReference = portal,
            ExpiresAt = DateTime.UtcNow.Add(timeout)
        };
        
        foreach (var channelType in channels)
        {
            try
            {
                var channel = _factory.GetChannel(channelType);
                var code = await channel.GetOtpAsync(otpRequest, timeout, context.CancellationToken);
                
                otpRequest.Status = OtpRequestStatus.Provided;
                otpRequest.ProvidedAt = DateTime.UtcNow;
                // Encrypt code before storing
                
                _db.OtpRequests.Add(otpRequest);
                await _db.SaveChangesAsync();
                
                context.SetOutput("code", code);
                return ActivityResult.Success;
            }
            catch (TimeoutException) when (channelType != channels.Last())
            {
                _logger.LogWarning("OTP channel {Channel} timeout, trying next", channelType);
                otpRequest.Channel = channels[channels.IndexOf(channelType) + 1];
                continue;
            }
            catch (TimeoutException)
            {
                otpRequest.Status = OtpRequestStatus.Expired;
                await _db.SaveChangesAsync();
                return ActivityResult.BusinessException("OTP timeout after all channels attempted");
            }
        }

        return ActivityResult.Success;
    }
}
```

```csharp
public class UiSpyHostedService : BackgroundService
{
    private readonly SapGuiElementDetector _detector;
    private readonly HubConnection _hubConnection;
    private readonly ILogger<UiSpyHostedService> _logger;

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("UI Spy service started");
        
        // Poll for element under cursor every 200ms
        while (!stoppingToken.IsCancellationRequested)
        {
            var elem = _detector.DetectElementAtCursor();
            if (elem != null)
            {
                // Send to Studio via SignalR
                await _hubConnection.InvokeAsync("ReceiveDetectedElement", elem.ElementId, elem.Type, elem.Text, elem.Position);
            }
            
            await Task.Delay(200, stoppingToken);
        }
    }
}
```
